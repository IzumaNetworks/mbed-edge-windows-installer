cmake_minimum_required(VERSION 3.20)
project(edge_msi_actions LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The MSI custom action must be built for native Windows x64")
endif()
configure_file(config.h.in config.h @ONLY)
add_library(edge_msi_actions SHARED actions.cpp)
target_include_directories(edge_msi_actions PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(edge_msi_actions PRIVATE cxx_std_17)
target_compile_definitions(edge_msi_actions PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(edge_msi_actions PRIVATE /W4 /WX /EHsc)
target_link_libraries(edge_msi_actions PRIVATE msi crypt32 shell32 ole32)
set_property(TARGET edge_msi_actions PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
